function searchText(q){
  q=norm(q);
  if(!q)return [];
  const all=uniqueItems();
  return all.map(x=>{
    const n=norm(x.name), lab=norm(x.lab), loc=norm(x.location);
    let score=0;
    if(n===q)score+=100;
    if(n.includes(q))score+=65;
    if(q.includes(n))score+=45;
    if(lab.includes(q))score+=15;
    if(loc.includes(q))score+=8;
    const words=q.match(/[가-힣a-z0-9]+/g)||[];
    words.forEach(w=>{if(n.includes(w))score+=12;});
    return {...x,score};
  }).filter(x=>x.score>0).sort((a,b)=>b.score-a.score).slice(0,30);
}

function handleImage(file){
  if(!file)return;
  const url=URL.createObjectURL(file);$("preview").src=url;$("preview").style.display="block";
  $("aiMessage").textContent="AI 이미지 분석 중…";
  if(state.model){
    state.model.classify($("preview")).then(preds=>{
      const labels=preds.map(p=>p.className.toLowerCase());
      const all=uniqueItems();
      let arr=all.map(x=>{
        const n=norm(x.name), meta=curatedFor(x);
        let score=0;
        labels.forEach((label,idx)=>{
          const l=norm(label);
          if(n.includes(l)||l.includes(n))score+=80-idx*10;
          if(meta.aliases.some(a=>l.includes(norm(a))||norm(a).includes(l)))score+=75-idx*8;
          if((label.includes("microscope")&&n.includes("현미경"))||(label.includes("lens")&&n.includes("렌즈")))score+=70;
        });
        return {...x,aiScore:score};
      }).filter(x=>x.aiScore>0).sort((a,b)=>b.aiScore-a.aiScore).slice(0,6);
      if(!arr.length) arr=all.slice(0,6).map((x,i)=>({...x,aiScore:60-i*2}));
      $("aiMessage").textContent="AI가 이미지 특징을 분석해 학교 물품 DB와 비교했습니다. 아래는 후보 목록입니다.";
      renderResults(arr,"results",true);
    }).catch(()=>fallbackImage());
  } else fallbackImage();
}
function fallbackImage(){
  const arr=uniqueItems().slice(0,6).map((x,i)=>({...x,aiScore:60-i*2}));
  $("aiMessage").textContent="브라우저 AI 모델을 불러오지 못해 학교 물품 DB 기반 후보를 표시합니다. 실제 배포에서는 학교 전용 이미지 분류 모델을 연결하는 것을 권장합니다.";
  renderResults(arr,"results",true);
}
